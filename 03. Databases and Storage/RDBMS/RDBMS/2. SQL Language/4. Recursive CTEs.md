# CTE Recursive


https://docs.arenadata.io/ru/ADPG/current/how-to/queries/common-table-expressions.html

ORDER BY is not allowed



```sql
with RECURSIVE cash_flow_calc as (
    select date, cash_flow, cash_flow as calc, 1 as lvl 
    from transactions 
    where id = 1
    UNION
    select t.date, t.cash_flow, t2.calc + t.cash_flow as calc, t2.lvl 
    from transactions t
    join (
        select t1.date, t1.cash_flow, c1.calc as calc, c1.lvl + 1 as lvl
        from cash_flow_calc c1
        inner join transactions t1 on t1.date > c1.date
        ORDER BY t1.date asc 
        limit 1
    ) t2 on t2.date = t.date
)
SELECT * from cash_flow_calc;
```


```sql
WITH RECURSIVE fib AS (
   SELECT 1 AS n, 1 AS fibn, 1 AS fibn1
   UNION ALL
      SELECT n+1,
             fibn1,
             fibn + fibn1
      FROM fib
)
SELECT n, fibn FROM fib
LIMIT 20;
```
