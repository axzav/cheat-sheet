# Authenticator

`Authenticator` наследует `AbstractAuthenticator` или реализует `AuthenticatorInterface`. 

```php
use Symfony\Component\Security\Http\Authenticator\AuthenticatorInterface;
use Symfony\Component\Security\Http\Authenticator\AbstractAuthenticator;

use Symfony\Component\HttpFoundation\Request;

use Symfony\Component\Security\Http\Authenticator\Passport\Passport;
use Symfony\Component\Security\Http\Authenticator\Passport\SelfValidatingPassport;

use Symfony\Component\Security\Http\Authenticator\Passport\Badge\UserBadge;

use Symfony\Component\Security\Core\Authentication\Token\TokenInterface;
use Symfony\Component\Security\Core\Exception\BadCredentialsException;

class ExampleAuthenticator implements AuthenticatorInterface
{
    public function supports(Request $request): ?bool
    {
        // Можно проверить кастомный хедер
        return $request->headers->has('auth-token');
        
        // Или куку
        return $request->cookies->has('TOKEN');
         
        // Или что запрос пришел из формы
        return $request->isMethod('POST')
            && $this->httpUtils->checkRequestPath($request, $this->options['check_path'])
            && 'form' === $request->getContentTypeFormat();
    }

    // authenticate делает проверки на основе кредов и возвращает Passport
    public function authenticate(Request $request): Passport
    {
        $tokenString = $request->headers->has('auth-token');
        $token = $this->tokenStorage->fetch($tokenString);

        // Можно бросить исключение если логика аутентификации не выполняется
        if (null === $token) {
            throw new BadCredentialsException('Invalid token');
        }

        $userIdentifier = $token->getUserIdentifier();

        // Если пароль пользователя не используется, то можно использовать SelfValidatingPassport, он не требует Credentials
        $passport = new SelfValidatingPassport(
            new UserBadge($userIdentifier) // В UserBadge передается `property` которое настроено в `provider` файервола
        );

        // Можно передать кастомную функцию загрузки пользователя вместо настройки `provider` файервола
        $passport =  new SelfValidatingPassport(
            new UserBadge($userIdentifier, function (string $token) use ($isSafe) {
                return $this->oAuth2->validateAccess($token, $isSafe);
            })
        );

        // Или так
        $userBadge = new UserBadge($userIdentifier, $this->userProvider->loadUserByIdentifier(...));

        // Если в аутентификаторе используется логин/пароль, то для их проверки нужно использовать Passport

        $passport = new Passport(
            new UserBadge($userIdentifier), 
            new PasswordCredentials($password)
        );

        // Можно передать CustomCredentials вместо PasswordCredentials
        $passport =  new Passport(
            new UserBadge($userIdentifier),
            new CustomCredentials(function($secret, $user) {
                return $user->getSecret() === $secret;
            }, $secret)
        );

        // Можно добавлять дополнительные Badge
        $passport->addBadge(new CsrfTokenBadge($this->options['csrf_token_id'], $credentials['csrf_token']));
        $passport->addBadge(new PasswordUpgradeBadge($credentials['password'], $this->userProvider));
    }

    // Если все проверки Badge прошли успешно, то создается токен и записывается в TokenStorage
    // Реализация этого метода есть в AbstractAuthenticator, проще наследоваться сразу от него
    public function createToken(Passport $passport, string $firewallName): TokenInterface
    {
        // Есть еще другие виды токенов (UsernamePasswordToken, PreAuthenticatedToken, RememberMeToken), для апи всегда подходит PostAuthenticationToken
        return new PostAuthenticationToken($passport->getUser(), $firewallName, $passport->getUser()->getRoles());
    }
    
    public function onAuthenticationSuccess(Request $request, TokenInterface $token, string $firewallName): ?Response
    {
        // Ничего не делать, запрос продолжит выполнение
        return null;
        
        // Или можно редиректнуть пользователя, если это логин форма
        return new RedirectResponse($this->urlGenerator->generate('some_route'));
    }

    public function onAuthenticationFailure(Request $request, AuthenticationException $exception): Response
    {
        // Ничего не делать, запрос продолжит выполнение пока не наткнется на Access Control
        return null;
        
        // Можно залогировать 
        if (null !== $this->logger) {
            if ($exception instanceof UserNotFoundException) {
                $this->logger->info('User for remember-me cookie not found.', ['exception' => $exception]);
            } elseif ($exception instanceof UnsupportedUserException) {
                $this->logger->warning('User class for remember-me cookie not supported.', ['exception' => $exception]);
            }
        }
        
        // Можно вернуть респонз
        return new Response(
            null,
            Response::HTTP_UNAUTHORIZED,
        );

        // Можно бросить исключение
        throw new AuthenticationException('Authentication Failed.');
    }
}
```
